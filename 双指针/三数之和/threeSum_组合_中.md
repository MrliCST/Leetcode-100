def backtrack(路径, 选择列表):
    if 满足结束条件:
        结果集.add(路径)
        return

    for 选择 in 选择列表:
        # 1. 做选择
        路径.append(选择)

        # 2. 递归进入下一层
        backtrack(路径, 新的选择列表)

        # 3. 撤销选择（回溯的关键！）
        路径.pop()
